```lua
local function startAuction(state: GameState, position: number, fromBankruptcy: boolean?)
	-- Everyone still in the game bids, starting with the player who declined. A bankrupt player's
	-- deeds are sold before the other debts are settled, so players who still owe money sit those out.
	local bidders = {}
	for offset = 0, #state.players - 1 do
		local player = state.players[(state.currentIndex - 1 + offset) % #state.players + 1]
		if not player.bankrupt and not (fromBankruptcy and hasDebt(state, player.id)) then
			table.insert(bidders, player.id)
		end
	end
	if #bidders == 0 then
		-- Nobody can bid: the deed stays with the bank.
		emit(state, { type = "AuctionEnded", position = position, winner = nil, amount = 0 })
		if fromBankruptcy then
			resume(state)
		else
			finishMove(state)
		end
		return
	end
	state.auction = { position = position, highBid = 0, highBidder = nil, bidders = bidders, nextBidder = 1, fromBankruptcy = fromBankruptcy == true }
	state.phase = "Auction"
	emit(state, { type = "AuctionStarted", position = position, bidders = table.clone(bidders) })
end
```

local function interestDue(state: Types.GameState, positions: { number }): number
	local total = 0

	end

```lua
--!strict
-- A per-key rate limit: at most one event per `seconds` for each key. Pure: the caller passes the time.

export type Throttle = {
	seconds: number,
	last: { [string]: number }, -- when each key last got through
}

local Throttle = {}

function Throttle.new(seconds: number): Throttle
	return { seconds = seconds, last = {} }
end

function Throttle.allow(throttle: Throttle, key: string, now: number): boolean
	local last = throttle.last[key]
	if last and now - last < throttle.seconds then
		return false
	end
	throttle.last[key] = now
	return true
end

function Throttle.forget(throttle: Throttle, key: string)
	throttle.last[key] = nil
end

return Throttle
```
